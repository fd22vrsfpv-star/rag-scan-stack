# Dependencies required to COLLECT and run the test suite.
#
# Why this file exists: the documented command was
#   pip install -q pytest pyyaml; pytest tests/
# which collected 2528 tests, of which 107 ERRORED on import — 103 on `fastapi`
# and 87 on `psycopg2`. Those looked like failures but were missing packages, and
# they hid 402 further tests that never collected at all (2930 collect with this
# file installed). A suite whose true state depends on an undocumented, ad-hoc
# pip line cannot honour the "keep the suite green" rule, because nobody can see
# what green is.
#
#   docker run --rm -v "$PWD":/repo -w /repo python:3.12-slim sh -c \
#     "pip install -q -r tests/requirements.txt && python -m pytest tests/ -q"
#
# Anything heavier than this (playwright browsers, a live database, a running
# stack) must be an importorskip / skipif inside the test, never a hard import —
# a missing optional dep should SKIP one file, not abort collection for all.

pytest>=8.0

# Parsers and knowledge YAML (extractors, rules, profiles).
PyYAML>=6.0

# Service modules under test declare FastAPI apps and Pydantic models at import
# time, so collection needs them even for pure-logic tests.
fastapi>=0.110
pydantic>=2.0

# db_utils and the ETL writers import psycopg2 at module scope. The binary wheel
# avoids needing libpq headers in the test image.
psycopg2-binary>=2.9

# HTTP clients used by dispatchers and by the live endpoint sweeps.
httpx>=0.27
requests>=2.31

# HTML parsing. The scan-recommender image ships it, so tests exercising the
# URL-fetch/extraction path importorskip on it — without it here, 17 tests
# report "beautifulsoup4 required" and silently do not run.
beautifulsoup4>=4.12

# Same class as beautifulsoup4 above: each is a light, pure-Python dep that a
# service image ships, so its tests importorskip and silently do not run here.
msgpack>=1.0        # msf_client / exploit_runner RPC
feedparser>=6.0     # news ingestion parser
pyflakes>=3.0       # the lint-gate test shells out to it
pytest-asyncio>=0.23  # async tests ERROR->skip without the plugin

# FastAPI needs this to BUILD a route that declares Form(...)/UploadFile — the
# check runs when the app is constructed, not when the route is called, so its
# absence raises RuntimeError("Form data requires python-multipart") out of
# fixture setup and reports as an ERROR, not a skip. Three tests errored this
# way once a database made them get far enough to construct the app.
python-multipart>=0.0.9

# The agent orchestrator imports langgraph at module scope, so the
# tests/test_langgraph_*.py modules skipped with "langgraph not importable here".
#
# MEASURED COST, both halves — the install is the cheap half and quoting only it
# understates this by ~28x:
#   pip install   6s -> 11s      (+5s, +22 packages, site-packages 114MB)
#   FULL SUITE  477s -> 617s     (+140s, +29% wall clock)
#   recovered   2743 -> 2771 passed, 438 -> 422 skipped
# The langgraph tests are ~139s of that: they compile a real StateGraph and drive
# it through checkpointed resumes, which is what makes them worth having.
#
# This file's own rule says anything heavier belongs behind an importorskip, and
# +29% is not obviously on the right side of it. It is here because those tests
# cover the exploit APPROVAL path — and the first run after adding it failed,
# exposing a test that had asserted a contract the engine stopped emitting when
# approval became multi-exploit. That defect was invisible for as long as the
# module skipped. Re-measure before adding the next heavy dep; do not treat this
# as precedent for "one more won't hurt".
langgraph>=0.2
